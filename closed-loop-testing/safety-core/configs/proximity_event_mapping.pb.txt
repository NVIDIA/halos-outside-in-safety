# Proximity mapping for SCENARIO=warehouse_20x20_1fl (PSF_APP=pxc), mounted over
# the image's /opt/nvidia/psf/apps/proximity/proximity_event_mapping.pb.txt.
#
# This scene has one Forklift and three walking workers. The image's own
# Forklift x Person rules stop at 2.0 / 1.0 m; the tiers below fit this truck.
#
# Output events are the three configs/pxc_sdm.conf binds to the forklift_person
# group (EVENT_12/13/14 -> CMD_NORMAL/CMD_REDUCE/CMD_STOP). EVENT_0..EVENT_11
# are reserved for ATL. The event id is the only channel the SDM has: it does
# not see the object classes.
#
# Tiers, centre to centre, measured from the 3D BEV box centres:
#   d <= 2.0 m        STOP     the forklift's front is ~1.3 m from its centre,
#                              so 2 m leaves ~0.7 m in front of the forks
#   2.0 < d <= 3.5 m  REDUCE   at 1.5 m/s the truck covers the 1.5 m band in
#                              about one perception round trip, and arrives at
#                              the STOP tier already slowed
#   d > 3.5 m         NORMAL
#
# Every distance_threshold_meters must stay <= the behavior-analytics
# proximityDetectionThreshold (4 m in the 3D app config). Above it,
# mdx-client drops the WHOLE rule group, STOP included, and logs nothing.
#
# File order does not matter: for each pair the rule with the smallest
# distance_threshold_meters at or above the measured distance is selected.

rules {
  name: "Forklift/Person no violation (> 3.5 m or SD false)"
  message_source: "mdx-frames"
  alert_type: "social_distancing"
  rule_id: "forklift_person_no_violation"
  object_type_primary: "Forklift"
  object_type_secondary: "Person"
  proximity_violation: false
  output_event: "EVENT_12"
}
rules {
  name: "Forklift/Person 3.5 m >= distance > 2 m"
  message_source: "mdx-frames"
  alert_type: "social_distancing"
  rule_id: "forklift_person_reduce"
  object_type_primary: "Forklift"
  object_type_secondary: "Person"
  proximity_violation: true
  distance_threshold_meters: 3.5
  output_event: "EVENT_13"
}
rules {
  name: "Forklift/Person distance <= 2 m"
  message_source: "mdx-frames"
  alert_type: "social_distancing"
  rule_id: "forklift_person_stop"
  object_type_primary: "Forklift"
  object_type_secondary: "Person"
  proximity_violation: true
  distance_threshold_meters: 2.0
  output_event: "EVENT_14"
}
